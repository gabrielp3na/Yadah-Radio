async function startNavigationMatch() { if (!pendingDestination || !myLastCoords) return; try { lblRouteStats.textContent = 'Calculando ruta...'; const url = `https://router.project-osrm.org/route/v1/driving/${myLastCoords.lon},${myLastCoords.lat};${pendingDestination.lon},${pendingDestination.lat}?overview=full&steps=true&geometries=geojson`; const response = await fetch(url); const data = await response.json(); if (data.code !== 'Ok' || !data.routes || !data.routes.length) { alert('No se encontró una ruta.'); return; } const route = data.routes[0]; const kilometers = (route.distance / 1000).toFixed(1); const minutes = Math.max(1, Math.round(route.duration / 60)); lblRouteStats.textContent = `🚀 ${pendingDestination.label} · ${kilometers} km · ${minutes} min`; navRouteInfo.style.display = 'flex'; removeActiveRoute(); activeRouteLayerId = 'yadah-route-layer'; activeRouteSourceId = 'yadah-route-source'; map.addSource(activeRouteSourceId, { type: 'geojson', data: { type: 'Feature', properties: {}, geometry: route.geometry } }); map.addLayer({ id: activeRouteLayerId, type: 'line', source: activeRouteSourceId, layout: { 'line-join': 'round', 'line-cap': 'round' }, paint: { 'line-color': '#38bdf8', 'line-width': 7, 'line-opacity': 0.9 } }); // Activar navegación antes de mover la cámara isNavigating = true; userMovedMap = false; firstPersonMode = true; updateViewButtonUI(); navSearchBar.style.display = 'none'; // Centrar el vehículo map.easeTo({ center: [myLastCoords.lon, myLastCoords.lat], zoom: 18.2, pitch: 62, bearing: lastValidHeading, offset: [0, 0], duration: 1000, essential: true }); } catch (error) { console.error('Error al iniciar navegación:', error); alert('No se pudo calcular la ruta.'); } }
